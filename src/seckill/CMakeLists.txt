find_package(OpenSSL REQUIRED)
find_package(PNG REQUIRED)
find_package(Threads REQUIRED)
find_package(nlohmann_json 3.11 REQUIRED)
find_path(HIREDIS_INCLUDE_DIR hiredis/hiredis.h REQUIRED)
find_library(HIREDIS_LIBRARY hiredis REQUIRED)
add_library(seckill_core redis.cpp captcha.cpp relay.cpp)
target_include_directories(seckill_core PUBLIC include ${HIREDIS_INCLUDE_DIR} ${PROJECT_SOURCE_DIR}/src/mq/include)
target_compile_definitions(seckill_core PUBLIC MALL_DEFAULT_LUA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/lua")
target_link_libraries(seckill_core PUBLIC mall_contracts OpenSSL::Crypto PNG::PNG ${HIREDIS_LIBRARY} nlohmann_json::nlohmann_json Threads::Threads)
add_executable(seckill_service main.cpp service.cpp)
target_include_directories(seckill_service SYSTEM PRIVATE ${PROJECT_SOURCE_DIR}/third_party/cpp-httplib)
target_link_libraries(seckill_service PRIVATE seckill_core)
target_compile_options(seckill_service PRIVATE -Wall -Wextra)
