cmake_minimum_required(VERSION 3.20)
project(seckill_mall VERSION 0.1.0 LANGUAGES CXX)
option(MALL_WITH_ROCKETMQ "Build RocketMQ transport adapter" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(mall_contracts INTERFACE)
target_include_directories(mall_contracts INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src/seckill/include)
add_subdirectory(src/seckill)
add_subdirectory(src/mq)
include(CTest)
if(BUILD_TESTING)
  add_executable(contract_tests tests/contracts_test.cpp)
  target_link_libraries(contract_tests PRIVATE mall_contracts)
  add_test(NAME state_contracts COMMAND contract_tests)
  add_test(NAME seckill_configuration COMMAND seckill_service --check-config)
  add_test(NAME mq_configuration COMMAND mall_mq --check-config)
  add_executable(relay_probe tests/relay_probe.cpp)
  target_link_libraries(relay_probe PRIVATE seckill_core)
endif()
